Task Definitions
Crew Tasks — Sequential Dependency Chain 4 tasks
# Task Name Description Agent Output
T1
parse_bim_data
Read the uploaded GLB/JSON file and extract all room names, dimensions (m²), door widths, material tags, and floor-to-ceiling heights.
 bim_analyst spatial_json
T2
check_compliance
Using T1 output, validate every room against ADA door width (>=0.9m), corridor width (>=1.2m), and Turkish building code ceiling heights.
compliance_checker violations_list
T3
estimate_costs
For each detected material (marble, oak, concrete, glass), fetch price-per-m² and multiply by surface area from T1 to produce a cost breakdown.
 cost_estimator cost_breakdown
T4
write_review_report
Compile T1 + T2 + T3 outputs into a structured design review JSON. Write to /output/review_{project_id}.json for dashboard consumption.
 report_writer review_report.json

# tasks.py — VR Architecture CrewAI Tasks from crewai import Task from .agents import bim_analyst, compliance_checker, cost_estimator, report_writer def parse_bim_task(file_path: str) -> Task: return Task( description=f"""Read the BIM/GLB file at {file_path}. Extract: room names, dimensions (m2), door widths, ceiling heights, and material tags. Return as JSON.""", expected_output="JSON with rooms[], materials[], dimensions{}", agent=bim_analyst() ) def compliance_task(context) -> Task: return Task( description="""Using the BIM spatial data from the previous task, check every room for ADA violations: - Door width >= 0.9m - Corridor width >= 1.2m - Ceiling height >= 2.4m Flag each violation with severity: HIGH / MEDIUM / LOW.""", expected_output="List of compliance violations with severity and room ref", agent=compliance_checker(), context=[context] ) def cost_estimation_task(context) -> Task: return Task( description="""For each material in the BIM data, query the price API. Multiply price_per_m2 by surface_area to get total cost. Include: material name, unit price, area, subtotal, grand total.""", expected_output="Cost breakdown table with subtotals and grand total in USD", agent=cost_estimator(), context=[context] ) def report_task(project_id: str, contexts: list) -> Task: return Task( description=f"""Compile spatial analysis, compliance violations, and cost breakdown into one structured JSON report. Save to: output/review_{project_id}.json""", expected_output="Complete design review JSON file written to disk", agent=report_writer(), context=contexts, output_file=f"output/review_{project_id}.json" )
04
Kickoff Code & Workflow
⚡
Integration Point: The kickoff is triggered from the existing AiAssistantPage.tsx via a POST to /api/Crew/run-review. The .NET backend spawns the Python subprocess, and the result JSON is picked up by the AiInsightsPage to render the live report on the dashboard.
📄 src/crew/main.py
python
# main.py — VR Architecture Crew Kickoff from crewai import Crew, Process from .tasks import parse_bim_task, compliance_task, cost_estimation_task, report_task def run_design_review(project_id: str, bim_file_path: str) -> dict: """ Kickoff the VR Architecture design review crew. Called by the .NET backend when a new GLB is uploaded. """ # 1. Define tasks in dependency order t1_bim = parse_bim_task(bim_file_path) t2_compliance = compliance_task(context=t1_bim) t3_cost = cost_estimation_task(context=t1_bim) t4_report = report_task(project_id, contexts=[t1_bim, t2_compliance, t3_cost]) # 2. Assemble the crew crew = Crew( agents=[ t1_bim.agent, t2_compliance.agent, t3_cost.agent, t4_report.agent, ], tasks=[t1_bim, t2_compliance, t3_cost, t4_report], process=Process.sequential, verbose=True, memory=True, # agents share context embedder={ "provider": "google", "config": {"model": "models/text-embedding-004"} } ) # 3. Kick off and return result result = crew.kickoff(inputs={ "project_id": project_id, "bim_file": bim_file_path }) return {"status": "success", "output": result} if __name__ == "__main__": run_design_review( project_id="harbour-museum-v3", bim_file_path="data/harbourfront_museum.json" )
